### Context7 Supabase Integration Patterns
```sql
-- Example RLS Policy for expenses table
CREATE POLICY "Users can manage their own expenses"
ON expenses
FOR ALL
TO authenticated
USING (auth.uid() = user_id);

-- User profile initialization
CREATE OR REPLACE FUNCTION public.initialize_user_profile()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO profiles (id, email, created_at)
  VALUES (NEW.id, NEW.email, NOW());
  
  -- Create default categories for user
  INSERT INTO categories (name, user_id, is_default)
  VALUES 
    ('Food & Dining', NEW.id, true),
    ('Transportation', NEW.id, true),
    ('Shopping', NEW.id, true),
    ('Entertainment', NEW.id, true);
    
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
```
