```dart
// Updated authentication service implementation
class AuthenticationService {
  // Remove: signInAnonymously()
  
  // Add: Email/password authentication
  Future<Result<User>> signUpWithPassword({
    required String email,
    required String password,
    String? displayName,
  }) async {
    try {
      final response = await _supabase.auth.signUp(
        email: email,
        password: password,
        data: displayName != null ? {'display_name': displayName} : null,
      );
      
      if (response.user != null) {
        return Result.success(response.user!);
      } else {
        return Result.failure(AppException.create(
          message: 'Sign up failed: No user returned',
          errorCode: 'AUTH_SIGNUP_FAILED',
        ));
      }
    } on AuthException catch (e) {
      return Result.failure(AppException.create(
        message: 'Sign up failed: ${e.message}',
        errorCode: 'AUTH_SUPABASE_ERROR',
        technicalDetails: e.toString(),
      ));
    }
  }
  
  Future<Result<User>> signInWithPassword({
    required String email,
    required String password,
  }) async {
    try {
      final response = await _supabase.auth.signInWithPassword(
        email: email,
        password: password,
      );
      
      if (response.user != null) {
        return Result.success(response.user!);
      } else {
        return Result.failure(AppException.create(
          message: 'Sign in failed: No user returned',
          errorCode: 'AUTH_SIGNIN_FAILED',
        ));
      }
    } on AuthException catch (e) {
      return Result.failure(AppException.create(
        message: 'Sign in failed: ${e.message}',
        errorCode: 'AUTH_SUPABASE_ERROR',
        technicalDetails: e.toString(),
      ));
    }
  }
}
```
